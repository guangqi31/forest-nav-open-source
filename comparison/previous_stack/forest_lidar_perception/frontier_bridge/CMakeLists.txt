cmake_minimum_required(VERSION 3.16)
project(forest_frontier_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav_msgs std_srvs geometry_msgs nav2_costmap_2d tf2_ros tf2)
  find_package(${dep} REQUIRED)
endforeach()
# 直接编译上游原文件，保留其算法及许可证；不启动上游自动发目标的 explore 节点。
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/m-explore-ros2/explore)
add_executable(frontier_bridge main.cpp ${UPSTREAM}/src/frontier_search.cpp)
target_include_directories(frontier_bridge PRIVATE ${UPSTREAM}/include)
ament_target_dependencies(frontier_bridge rclcpp nav_msgs std_srvs geometry_msgs nav2_costmap_2d tf2_ros tf2)
install(TARGETS frontier_bridge DESTINATION lib/${PROJECT_NAME})
install(FILES ${UPSTREAM}/../LICENSE DESTINATION share/${PROJECT_NAME} RENAME UPSTREAM_LICENSE)
ament_package()
